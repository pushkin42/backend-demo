BEGIN
  DECLARE done INT DEFAULT 0;
  DECLARE item_id INT;
  DECLARE cur CURSOR FOR SELECT catalog_items.id FROM catalog_items WHERE NAME LIKE CONCAT('%',items_like,'%');
  DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
Open cur;
WHILE not done  DO 
FETCH cur INTO item_id;
INSERT INTO catalog_category_item_pivot(catalog_category_id, catalog_item_id) VALUES (category_id, item_id);
END WHILE;
Close cur; 
END

BEGIN
    START TRANSACTION;

    -- 1. Сложить quantity для совпадающих товаров
    UPDATE cart_items tgt
    JOIN cart_items src
      ON src.catalog_item_id = tgt.catalog_item_id
     AND (
            (from_type = 'session' AND src.session_id = from_id)
         OR (from_type = 'user'    AND src.user_id    = from_id)
         )
     AND (
            (to_type = 'session' AND tgt.session_id = to_id)
         OR (to_type = 'user'    AND tgt.user_id    = to_id)
         )
    SET tgt.quantity = tgt.quantity + src.quantity, tgt.cart_id=0;

    -- 2. Перенести уникальные позиции
    UPDATE cart_items src
    LEFT JOIN cart_items tgt
      ON tgt.catalog_item_id = src.catalog_item_id
     AND (
            (to_type = 'session' AND tgt.session_id = to_id)
         OR (to_type = 'user'    AND tgt.user_id    = to_id)
         )
    SET
        src.user_id    = IF(to_type='user', to_id, NULL),
        src.session_id= IF(to_type='session', to_id, NULL),
        src.cart_id=0
    WHERE
        tgt.id IS NULL
        AND (
            (from_type='session' AND src.session_id = from_id)
         OR (from_type='user'    AND src.user_id    = from_id)
        );

    -- 3. Удалить дубликаты источника
    DELETE src
    FROM cart_items src
    JOIN cart_items tgt
      ON tgt.catalog_item_id = src.catalog_item_id
     AND (
            (to_type = 'session' AND tgt.session_id = to_id)
         OR (to_type = 'user'    AND tgt.user_id    = to_id)
         )
    WHERE
        (
            (from_type='session' AND src.session_id = from_id)
         OR (from_type='user'    AND src.user_id    = from_id)
        );

    COMMIT;
END

WITH RECURSIVE `category_tree` AS (
SELECT `catalog_categories`.`id` AS `category_id`,`catalog_categories`.`id` AS `root_id`
FROM `catalog_categories` UNION ALL
SELECT `c`.`id` AS `category_id`,`ct`.`root_id` AS `root_id`
FROM (`catalog_categories` `c`
JOIN `category_tree` `ct` ON((`c`.`parent_id` = `ct`.`category_id`))))
SELECT `ct`.`root_id` AS `category_id`,`cp`.`catalog_item_id` AS `catalog_item_id`,`cp`.`catalog_category_id` AS `actual_category_id`
FROM (`category_tree` `ct`
JOIN `catalog_category_item_pivot` `cp` ON((`cp`.`catalog_category_id` = `ct`.`category_id`)))

SELECT ROW_NUMBER() OVER (
ORDER BY `t`.`interval_start`) AS `row_num`,`t`.`interval_start` AS `interval_start`,`t`.`interval_end` AS `interval_end`
FROM (
SELECT (CURDATE() + INTERVAL `intervals`.`minutes` MINUTE) AS `interval_start`,((CURDATE() + INTERVAL `intervals`.`minutes` MINUTE) + INTERVAL 5 MINUTE) AS `interval_end`
FROM (
SELECT ((`h`.`hour` * 60) + `m`.`minute`) AS `minutes`
FROM ((
SELECT 0 AS `hour` UNION
SELECT 1 AS `1` UNION
SELECT 2 AS `2` UNION
SELECT 3 AS `3` UNION
SELECT 4 AS `4` UNION
SELECT 5 AS `5` UNION
SELECT 6 AS `6` UNION
SELECT 7 AS `7` UNION
SELECT 8 AS `8` UNION
SELECT 9 AS `9` UNION
SELECT 10 AS `10` UNION
SELECT 11 AS `11` UNION
SELECT 12 AS `12` UNION
SELECT 13 AS `13` UNION
SELECT 14 AS `14` UNION
SELECT 15 AS `15` UNION
SELECT 16 AS `16` UNION
SELECT 17 AS `17` UNION
SELECT 18 AS `18` UNION
SELECT 19 AS `19` UNION
SELECT 20 AS `20` UNION
SELECT 21 AS `21` UNION
SELECT 22 AS `22` UNION
SELECT 23 AS `23`) `h`
JOIN (
SELECT 0 AS `minute` UNION
SELECT 5 AS `5` UNION
SELECT 10 AS `10` UNION
SELECT 15 AS `15` UNION
SELECT 20 AS `20` UNION
SELECT 25 AS `25` UNION
SELECT 30 AS `30` UNION
SELECT 35 AS `35` UNION
SELECT 40 AS `40` UNION
SELECT 45 AS `45` UNION
SELECT 50 AS `50` UNION
SELECT 55 AS `55`) `m`)
WHERE (((`h`.`hour` * 60) + `m`.`minute`) < 1440)) `intervals`
ORDER BY (CURDATE() + INTERVAL `intervals`.`minutes` MINUTE)) `t`

BEGIN
    DROP TEMPORARY TABLE IF EXISTS _agg;
    CREATE TEMPORARY TABLE _agg ENGINE = MEMORY
    SELECT 
        FLOOR(UNIX_TIMESTAMP(created_at) / 300) AS interval_num,
        ROUND(AVG(cpu_usage)) AS avg_cpu,
        MAX(cpu_usage) AS max_cpu,
        ROUND(AVG(current_ram_loading)) AS avg_ram,
        MAX(current_ram_loading) AS max_ram,
        ROUND(AVG(network_rx), 2) AS avg_network_rx,
        MAX(network_rx) AS max_network_rx,
        ROUND(AVG(network_tx), 2) AS avg_network_tx,
        MAX(network_tx) AS max_network_tx,
        AVG(current_network_loading) AS avg_network_loading,
        AVG(last_ping) AS avg_last_ping
    FROM remote_server_stats 
    WHERE remote_server_id = p_server_id
      AND created_at >= CURDATE()
      AND created_at < CURDATE() + INTERVAL 1 DAY
    GROUP BY interval_num;
    
    ALTER TABLE _agg ADD PRIMARY KEY (interval_num);
    
    SELECT 
        n.n + 1 AS row_num,
        CURDATE() + INTERVAL (n.n * 5) MINUTE AS interval_start,
        CURDATE() + INTERVAL ((n.n + 1) * 5) MINUTE AS interval_end,
        COALESCE(a.avg_cpu, 0) AS avg_cpu,
        COALESCE(a.max_cpu, 0) AS max_cpu,
        COALESCE(a.avg_ram, 0) AS avg_ram,
        COALESCE(a.max_ram, 0) AS max_ram,
        COALESCE(a.avg_network_rx, 0) AS avg_network_rx,
        COALESCE(a.max_network_rx, 0) AS max_network_rx,
        COALESCE(a.avg_network_tx, 0) AS avg_network_tx,
        COALESCE(a.max_network_tx, 0) AS max_network_tx,
        COALESCE(a.avg_network_loading, 0) AS avg_network_loading,
        COALESCE(a.avg_last_ping, 0) AS avg_last_ping
		  
    FROM (
        WITH RECURSIVE nums(n) AS (SELECT 0 UNION ALL SELECT n+1 FROM nums WHERE n < 287)
        SELECT n FROM nums
    ) n
    LEFT JOIN _agg a ON a.interval_num = FLOOR(UNIX_TIMESTAMP(CURDATE()) / 300) + n.n
    ORDER BY n.n;
    
    DROP TEMPORARY TABLE _agg;
END

CREATE TABLE `statistic_entries` (
	`id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
	`ip` VARCHAR(20) NOT NULL COMMENT 'IP адрес клиента' COLLATE 'utf8mb4_unicode_ci',
	`hostname` VARCHAR(255) NULL DEFAULT NULL COMMENT 'Имя хоста клиента' COLLATE 'utf8mb4_unicode_ci',
	`cookies` JSON NULL DEFAULT NULL COMMENT 'Заголовок cookie из запроса',
	`lang` JSON NULL DEFAULT NULL COMMENT 'Заголовок accept-language из запроса',
	`encoding` JSON NULL DEFAULT NULL COMMENT 'Заголовок accept-encoding из запроса',
	`referer` TEXT NULL DEFAULT NULL COMMENT 'Заголовок referer из запроса' COLLATE 'utf8mb4_unicode_ci',
	`host` VARCHAR(255) NULL DEFAULT NULL COMMENT 'Заголовок host из запроса' COLLATE 'utf8mb4_unicode_ci',
	`method` VARCHAR(20) NULL DEFAULT NULL COMMENT 'Метод запроса' COLLATE 'utf8mb4_unicode_ci',
	`platform` VARCHAR(255) NULL DEFAULT NULL COMMENT 'Платформа клиента (windows/linux/etc)' COLLATE 'utf8mb4_unicode_ci',
	`is_mobile` TINYINT(1) NULL DEFAULT NULL COMMENT 'Мобильное устройство',
	`is_bot` TINYINT(1) NULL DEFAULT NULL COMMENT 'Это бот',
	`ua` VARCHAR(2048) NULL DEFAULT NULL COMMENT 'User agent клиента' COLLATE 'utf8mb4_unicode_ci',
	`dnt` TINYINT(1) NULL DEFAULT NULL COMMENT 'Есть ли заголовок do not track',
	`location` JSON NULL DEFAULT NULL COMMENT 'Координаты устройства клиента (если доступно)',
	`city` VARCHAR(255) NULL DEFAULT NULL COMMENT 'Город клиента' COLLATE 'utf8mb4_unicode_ci',
	`query` TEXT NULL DEFAULT NULL COMMENT 'Строка запроса' COLLATE 'utf8mb4_unicode_ci',
	`user_id` BIGINT UNSIGNED NULL DEFAULT NULL COMMENT 'ID пользователя (если авторизован)',
	`session_id` VARCHAR(255) NULL DEFAULT NULL COMMENT 'ID сессии (внутреннее значение)' COLLATE 'utf8mb4_unicode_ci',
	`data` JSON NULL DEFAULT NULL COMMENT 'Дополнительные данные',
	`counter` BIGINT UNSIGNED NULL DEFAULT '0' COMMENT 'Счетчик повторов',
	`response_code` SMALLINT NULL DEFAULT NULL COMMENT 'Код ответа',
	`response_text` VARCHAR(700) NULL DEFAULT NULL COMMENT 'Текст ответа (если ошибка)' COLLATE 'utf8mb4_unicode_ci',
	`created_at` TIMESTAMP NULL DEFAULT NULL,
	`updated_at` TIMESTAMP NULL DEFAULT NULL,
	`deleted_at` TIMESTAMP NULL DEFAULT NULL,
	`country_code` VARCHAR(10) NULL DEFAULT NULL COMMENT 'Код страны (RU/EN...)' COLLATE 'utf8mb4_unicode_ci',
	`country` VARCHAR(255) NULL DEFAULT NULL COMMENT 'Страна (если определено)' COLLATE 'utf8mb4_unicode_ci',
	`bl_data_honeypot` JSON NULL DEFAULT NULL COMMENT 'Данные DNS honeypot project',
	`bl_data_abuseip` JSON NULL DEFAULT NULL COMMENT 'Данные ABUSEIPDB',
	`usage_type` VARCHAR(255) NULL DEFAULT NULL COMMENT 'Класс хоста (по версии ABUSEIPDB)' COLLATE 'utf8mb4_unicode_ci',
	`wrong_host` TINYINT(1) NULL DEFAULT '0' COMMENT 'Хост не совпадает с сайтом',
	`score` INT NULL DEFAULT '0' COMMENT 'Индекс "вредности"',
	`report_count` INT NULL DEFAULT '0' COMMENT 'Количество жалоб',
	`last_reported_at` TIMESTAMP NULL DEFAULT NULL COMMENT 'Дата и время последней жалобы',
	`whitelisted` TINYINT NULL DEFAULT NULL COMMENT 'В белом списке abuseip',
	`query_hash_sha1_binary` VARBINARY(20) AS (unhex(sha(`query`))) stored COMMENT 'Хеш sha1 для url (в HEX виде)',
	PRIMARY KEY (`id`) USING BTREE,
	INDEX `statistic_entries_user_id_foreign` (`user_id`) USING BTREE,
	INDEX `ip` (`ip`) USING BTREE,
	INDEX `hostname` (`hostname`) USING BTREE,
	INDEX `session_id` (`session_id`) USING BTREE,
	INDEX `counter` (`counter`) USING BTREE,
	INDEX `response_text` (`response_text`) USING BTREE,
	INDEX `response_code` (`response_code`) USING BTREE,
	INDEX `city` (`city`) USING BTREE,
	INDEX `platform` (`platform`) USING BTREE,
	INDEX `host` (`host`) USING BTREE,
	INDEX `statistic_entries_country_code_index` (`country_code`) USING BTREE,
	INDEX `statistic_entries_country_index` (`country`) USING BTREE,
	INDEX `statistic_entries_usage_type_index` (`usage_type`) USING BTREE,
	INDEX `statistic_entries_wrong_host_index` (`wrong_host`) USING BTREE,
	INDEX `statistic_entries_score_index` (`score`) USING BTREE,
	INDEX `statistic_entries_report_count_index` (`report_count`) USING BTREE,
	INDEX `statistic_entries_last_reported_at_index` (`last_reported_at`) USING BTREE,
	INDEX `idx_query_hash_deleted_counter` (`deleted_at`, `counter`) USING BTREE,
	INDEX `idx_perfect` (`deleted_at`, `counter`, `id`) USING BTREE,
	INDEX `counter_deleted_at_query_hash_sha1_binary` (`query_hash_sha1_binary`, `deleted_at`, `counter`) USING BTREE,
	FULLTEXT INDEX `referer` (`referer`),
	FULLTEXT INDEX `query` (`query`),
	CONSTRAINT `statistic_entries_user_id_foreign` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON UPDATE CASCADE ON DELETE SET NULL
)
COMMENT='Таблица для сбора статистики посещений'
COLLATE='utf8mb4_unicode_ci'
ENGINE=InnoDB;
